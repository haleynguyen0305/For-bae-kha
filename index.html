music.addEventListener("timeupdate",()=>{ if( music.duration && music.currentTime >= music.duration - 3 && !loopingMusic && !music.muted ){ loopingMusic = true; const fadeSteps = 30; const startVolume = music.volume; let step = 0; const fadeOut = setInterval(()=>{ step++; music.volume = Math.max( 0, startVolume * (1 - step / fadeSteps) ); if(step >= fadeSteps){ clearInterval(fadeOut); music.currentTime = 0; music.volume = 0; let fadeStep = 0; const fadeIn = setInterval(()=>{ fadeStep++; music.volume = Math.min( 0.75, 0.75 * (fadeStep / fadeSteps) ); if(fadeStep >= fadeSteps){ clearInterval(fadeIn); loopingMusic = false; } }, 70); } }, 70); } });
